{ 原版抄底指标 — 边沿确认版（交易用，非看盘留字） }
{ 逻辑与原版一致: AAA/BBB → CCC → JJJ → DXDX 为「抄底」边沿 }
{ 全部基于已收盘数据；DXDX 为边沿，避免字样长期不消失被当成持续信号 }

D:=EMA(CLOSE,12)-EMA(CLOSE,26);
A:=EMA(D,9);
M:=(D-A)*2;

N1:=BARSLAST((REF(M,1)>=0) AND (M<0));
MM1:=BARSLAST((REF(M,1)<=0) AND (M>0));

CC1:=LLV(CLOSE,(N1+1));
CC2:=REF(CC1,(MM1+1));
CC3:=REF(CC2,(MM1+1));
DIFL1:=LLV(D,(N1+1));
DIFL2:=REF(DIFL1,(MM1+1));
DIFL3:=REF(DIFL2,(MM1+1));

CH1:=HHV(CLOSE,(MM1+1));
CH2:=REF(CH1,(N1+1));
CH3:=REF(CH2,(N1+1));
DIFH1:=HHV(D,(MM1+1));
DIFH2:=REF(DIFH1,(N1+1));
DIFH3:=REF(DIFH2,(N1+1));

AAA:=(CC1<CC2) AND (DIFL1>DIFL2) AND (REF(M,1)<0) AND (D<0);
BBB:=(CC1<CC3) AND (DIFL1<DIFL2) AND (DIFL1>DIFL3) AND (REF(M,1)<0) AND (D<0);
CCC:=(AAA OR BBB) AND (D<0);

JJJ:=REF(CCC,1) AND (ABS(REF(D,1))>=ABS(D)*1.01);
DXDX:=(REF(JJJ,1)=0) AND JJJ;

{ 顶背离卖出边沿（回测未作为出场主规则，仅保留可对照） }
ZJDBL:=(CH1>CH2) AND (DIFH1<DIFH2) AND (REF(M,1)>0) AND (D>0);
GXDBL:=(CH1>CH3) AND (DIFH1>DIFH2) AND (DIFH1<DIFH3) AND (REF(M,1)>0) AND (D>0);
DBBL:=(ZJDBL OR GXDBL) AND (D>0);
DBJG:=REF(DBBL,1) AND (REF(D,1)>=D*1.01);
DBJGXC:=REF(NOT(DBJG),1) AND DBJG;

DRAWTEXT(DXDX,D/0.81,'抄底'),COLORRED;
DRAWTEXT(DBJGXC,D*1.21,'卖出'),COLORGREEN;
